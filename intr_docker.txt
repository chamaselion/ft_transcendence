DOCKER:
docker ps (list containers)
docker ps -a (all containers which off too)
docker start <container>
docker restart <container>
docker rm <container>
docker logs <container>
docker logs -f <container> (logs in RT)
docker exec -it <container> bash

Images:
docker images
docker pull <image>
docker build -t myimage .
docker rmi <image>

Network:
docker network ls
docker network inspect <network>
docker network rm <network>

Volumes:
docker volume ls
docker volume inspect <volume>
docker volume rm <volume>
docker volume prune

Cleaning:
docker system df           # сколько места занимает Docker
docker system prune        # удалить неиспользуемые объекты
docker system prune -a     # удалить еще и неиспользуемые образы


Docker Compose
Запуск:
docker compose up
docker compose up -d
docker compose down
docker compose down -v     # удалить тома
docker compose restart

Пересборка:
docker compose up --build
docker compose build
docker compose build --no-cache

Просмотр:
docker compose ps
docker compose logs
docker compose logs -f
docker compose config

Работа с сервисами:
docker compose start
docker compose stop
docker compose restart
docker compose exec <service> bash
docker compose exec <service> sh

Например:
docker compose exec backend bash
docker compose exec db psql -U postgres
Остановка и удаление
docker compose down
docker compose down -v          # удалить volumes
docker compose down --rmi all   # удалить образы
